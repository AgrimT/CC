STEPS FOR VERCEL:
1. make index.html file and paste code 
2. make repo on github, upload index.html there
3. login to vercel with github
4. add new project on vercel
5. select the github repo
6. click deploy


STEPS FOR AWS AMPLIFY:
1. make a repository on git my-paas-app and paste index.html code
2. sign in into aws amplify and use root email.
3. use code in phone to login.
4. near search bar, click on squares then aws amplify
5. click create new app
6. host using git
7. select the repository
8. then just do next next then save and deploy
9. See the deployment

<script>
  // All rates relative to USD. Update these values as needed.
  const RATES = {
    USD: 1,
    EUR: 0.92,
    GBP: 0.79,
    NGN: 1590,
    JPY: 155.4,
    CAD: 1.37,
    AUD: 1.55,
    CHF: 0.90,
    CNY: 7.26,
    INR: 83.5,
    BRL: 5.12,
    MXN: 17.2,
    SGD: 1.35,
    HKD: 7.83,
    ZAR: 18.7,
    GHS: 15.6,
    KES: 129.5,
    EGP: 30.9,
    SAR: 3.75,
    AED: 3.67,
    NOK: 10.85,
    SEK: 10.52,
    TRY: 32.5,
    PKR: 278.5,
    BDT: 110.2,
  };

  const NAMES = {
    USD:'US Dollar',EUR:'Euro',GBP:'British Pound',NGN:'Nigerian Naira',
    JPY:'Japanese Yen',CAD:'Canadian Dollar',AUD:'Australian Dollar',
    CHF:'Swiss Franc',CNY:'Chinese Yuan',INR:'Indian Rupee',
    BRL:'Brazilian Real',MXN:'Mexican Peso',SGD:'Singapore Dollar',
    HKD:'Hong Kong Dollar',ZAR:'South African Rand',GHS:'Ghanaian Cedi',
    KES:'Kenyan Shilling',EGP:'Egyptian Pound',SAR:'Saudi Riyal',
    AED:'UAE Dirham',NOK:'Norwegian Krone',SEK:'Swedish Krona',
    TRY:'Turkish Lira',PKR:'Pakistani Rupee',BDT:'Bangladeshi Taka'
  };

  const POPULAR = [
    {pair:'EUR / USD', from:'EUR', to:'USD'},
    {pair:'GBP / USD', from:'GBP', to:'USD'},
    {pair:'USD / NGN', from:'USD', to:'NGN'},
    {pair:'USD / JPY', from:'USD', to:'JPY'},
    {pair:'USD / INR', from:'USD', to:'INR'},
    {pair:'USD / GHS', from:'USD', to:'GHS'},
  ];

  let convHistory = [];

  // Populate selects
  const fromSel = document.getElementById('from-cur');
  const toSel = document.getElementById('to-cur');
  Object.keys(RATES).forEach(c => {
    [fromSel, toSel].forEach(sel => {
      const o = document.createElement('option');
      o.value = c;
      o.textContent = `${c} — ${NAMES[c] || c}`;
      sel.appendChild(o);
    });
  });
  fromSel.value = 'USD';
  toSel.value = 'NGN';

  // Popular rates display
  document.getElementById('rates-grid').innerHTML = POPULAR.map(p => {
    const rate = (RATES[p.to] / RATES[p.from]).toFixed(4);
    return `<div class="rate-chip">
      <span class="pair">${p.pair}</span>
      <span class="val">${rate}</span>
    </div>`;
  }).join('');

  function convert() {
    document.getElementById('err').textContent = '';
    const amount = parseFloat(document.getElementById('amount').value);
    const from = fromSel.value;
    const to = toSel.value;

    if (isNaN(amount) || amount <= 0) {
      document.getElementById('err').textContent = 'Please enter a valid positive amount.';
      return;
    }

    const inUSD = amount / RATES[from];
    const result = inUSD * RATES[to];
    const rate = RATES[to] / RATES[from];

    const resultStr = result.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 4 });

    document.getElementById('result-box').style.display = 'block';
    document.getElementById('r-from').textContent = `${amount.toLocaleString()} ${from} (${NAMES[from]})`;
    document.getElementById('r-result').textContent = `${resultStr} ${to}`;
    document.getElementById('r-rate').textContent = `1 ${from} = ${rate.toFixed(4)} ${to}`;

    // History
    convHistory.unshift({ from, to, amount, result: resultStr });
    if (convHistory.length > 5) convHistory.pop();
    renderHistory();
  }

  function renderHistory() {
    const histEl = document.getElementById('history');
    const list = document.getElementById('history-list');
    histEl.style.display = 'block';
    list.innerHTML = convHistory.map(h =>
      `<div class="history-item">
        <span>${h.amount.toLocaleString()} ${h.from} → ${h.to}</span>
        <span>${h.result} ${h.to}</span>
      </div>`
    ).join('');
  }

  function swapCur() {
    const tmp = fromSel.value;
    fromSel.value = toSel.value;
    toSel.value = tmp;
  }
</script>
