STEPS FOR VERCEL:
1. make index.html file and paste code 
2. make repo on github, upload index.html there
3. login to vercel with github
4. add new project on vercel
5. select the github repo
6. click deploy


STEPS FOR AWS AMPLIFY:
1. make a repository on git my-paas-app and paste index.html code
2. sign in into aws amplify and use root email.
3. use code in phone to login.
4. near search bar, click on squares then aws amplify
5. click create new app
6. host using git
7. select the repository
8. then just do next next then save and deploy
9. See the deployment

  const topics = { general: "General Knowledge", science: "Science", math: "Mathematics", tech: "Technology" };
  let selected = 'general', questions = [], qIndex = 0, score = 0;
  let timerInterval, answered = false, answers = [];

  // Render topic buttons
  const topicSel = document.getElementById('topic-select');
  Object.entries(topics).forEach(([key, label]) => {
    const b = document.createElement('button');
    b.className = 'topic-btn' + (key === 'general' ? ' selected' : '');
    b.textContent = label;
    b.onclick = () => { selected = key; document.querySelectorAll('.topic-btn').forEach(x=>x.classList.remove('selected')); b.classList.add('selected'); };
    topicSel.appendChild(b);
  });

  function shuffle(arr) { return [...arr].sort(()=>Math.random()-0.5); }

  function startQuiz() {
    questions = shuffle(ALL_QUESTIONS[selected]).slice(0, 10);
    qIndex = 0; score = 0; answers = [];
    show('quiz-screen');
    loadQuestion();
  }

  function show(id) {
    document.querySelectorAll('.screen').forEach(s=>s.classList.remove('active'));
    document.getElementById(id).classList.add('active');
  }

  function loadQuestion() {
    clearInterval(timerInterval);
    answered = false;
    const q = questions[qIndex];
    document.getElementById('q-counter').textContent = `Question ${qIndex+1} of ${questions.length}`;
    document.getElementById('q-text').textContent = q.q;
    document.getElementById('prog-bar').style.width = `${((qIndex)/questions.length)*100}%`;
    document.getElementById('feedback').style.display = 'none';
    document.getElementById('next-btn').style.display = 'none';
    const letters = ['A','B','C','D'];
    const opts = document.getElementById('options');
    opts.innerHTML = q.o.map((o,i)=>`
      <button class="option-btn" onclick="selectAnswer(${i})">
        <span class="option-letter">${letters[i]}</span>${o}
      </button>`).join('');
    let t = 20;
    document.getElementById('timer').textContent = `${t}s`;
    timerInterval = setInterval(()=>{
      t--;
      document.getElementById('timer').textContent = `${t}s`;
      if(t<=0){ clearInterval(timerInterval); if(!answered) autoFail(); }
    }, 1000);
  }

  function selectAnswer(i) {
    if (answered) return;
    answered = true;
    clearInterval(timerInterval);
    const q = questions[qIndex];
    const btns = document.querySelectorAll('.option-btn');
    btns.forEach(b=>b.disabled=true);
    const fb = document.getElementById('feedback');
    if (i === q.a) {
      score++;
      btns[i].classList.add('correct');
      fb.className='feedback correct'; fb.textContent='Correct!';
    } else {
      btns[i].classList.add('wrong');
      btns[q.a].classList.add('correct');
      fb.className='feedback wrong'; fb.textContent=`Wrong. The correct answer is "${q.o[q.a]}".`;
    }
    fb.style.display='block';
    answers.push({ q: q.q, your: q.o[i], correct: q.o[q.a], right: i===q.a });
    document.getElementById('next-btn').style.display='block';
  }

  function autoFail() {
    answered = true;
    const q = questions[qIndex];
    const btns = document.querySelectorAll('.option-btn');
    btns.forEach(b=>b.disabled=true);
    btns[q.a].classList.add('correct');
    const fb = document.getElementById('feedback');
    fb.className='feedback wrong'; fb.textContent=`Time's up! The answer was "${q.o[q.a]}".`;
    fb.style.display='block';
    answers.push({ q: q.q, your: 'No answer', correct: q.o[q.a], right: false });
    document.getElementById('next-btn').style.display='block';
  }

  function nextQuestion() {
    qIndex++;
    if (qIndex < questions.length) { loadQuestion(); }
    else { showResults(); }
  }

  function showResults() {
    show('result-screen');
    document.getElementById('score-num').textContent = score;
    document.getElementById('score-den').textContent = `/ ${questions.length}`;
    const pct = (score/questions.length)*100;
    document.getElementById('result-title').textContent = pct>=80?'Excellent!':pct>=60?'Good Job!':pct>=40?'Not Bad':'Keep Practicing';
    document.getElementById('result-msg').textContent = `You got ${score} out of ${questions.length} correct (${Math.round(pct)}%).`;
    document.getElementById('review').innerHTML = answers.map(a=>`
      <div class="review-item ${a.right?'correct-r':'wrong-r'}">
        <div class="rq">${a.q}</div>
        <div class="ra">Your answer: <strong>${a.your}</strong> ${a.right?'✓':'✗'}</div>
        ${!a.right?`<div class="ra">Correct: <strong>${a.correct}</strong></div>`:''}
      </div>`).join('');
  }

  function resetQuiz() { show('start-screen'); }
