STEPS FOR VERCEL:
1. make index.html file and paste code 
2. make repo on github, upload index.html there
3. login to vercel with github
4. add new project on vercel
5. select the github repo
6. click deploy


STEPS FOR AWS AMPLIFY:
1. make a repository on git my-paas-app and paste index.html code
2. sign in into aws amplify and use root email.
3. use code in phone to login.
4. near search bar, click on squares then aws amplify
5. click create new app
6. host using git
7. select the repository
8. then just do next next then save and deploy
9. See the deployment

<header>
  <h1>CGPA Calculator</h1>
  <p>Track your grades across all semesters</p>

<div class="container">
  <div id="semesters-container"></div>
  <div style="text-align:center; margin-bottom:24px;">
    <button class="btn btn-ghost" onclick="addSemester()">+ Add Semester</button>
  </div>
  <div class="card" id="results-card" style="display:none;">
    <div class="card-title">Cumulative Results</div>
    <div class="results" id="results-container"></div>
  </div>
  <div style="display:flex;justify-content:center;margin-top:8px;">
    <button class="btn btn-accent" onclick="calculate()">Calculate CGPA</button>
  </div>
</div>
<script>
  const gradePoints = {
    'A+':4.0,'A':4.0,'A-':3.7,'B+':3.3,'B':3.0,'B-':2.7,
    'C+':2.3,'C':2.0,'C-':1.7,'D+':1.3,'D':1.0,'F':0.0
  };
  let semCount = 0;
  let semesters = [];

  function addSemester() {
    semCount++;
    semesters.push({ id: semCount, courses: [{},{},{}] });
    render();
  }

  function removeSemester(id) {
    semesters = semesters.filter(s => s.id !== id);
    render();
  }

  function addCourse(id) {
    semesters.find(s => s.id === id).courses.push({});
    render();
  }

  function removeCourse(id, idx) {
    const s = semesters.find(s => s.id === id);
    if (s.courses.length > 1) { s.courses.splice(idx, 1); render(); }
  }

  function updateCourse(id, idx, field, val) {
    semesters.find(s => s.id === id).courses[idx][field] = val;
  }

  function render() {
    const c = document.getElementById('semesters-container');
    c.innerHTML = '';
    semesters.forEach((sem, si) => {
      const d = document.createElement('div');
      d.className = 'card';
      d.innerHTML = `
        <div class="semester-header">
          <span class="semester-title">Semester ${si + 1}</span>
          <button class="btn btn-ghost btn-sm" onclick="removeSemester(${sem.id})">Remove</button>
        </div>
        <div class="course-row header-row">
          <span>Course Name</span><span>Grade</span><span>Credits</span><span></span>
        </div>
        ${sem.courses.map((c, ci) => `
          <div class="course-row">
            <input type="text" placeholder="Course name" value="${c.name||''}" onchange="updateCourse(${sem.id},${ci},'name',this.value)"/>
            <select onchange="updateCourse(${sem.id},${ci},'grade',this.value)">
              <option value="">Grade</option>
              ${Object.keys(gradePoints).map(g=>`<option value="${g}" ${c.grade===g?'selected':''}>${g}</option>`).join('')}
            </select>
            <input type="number" min="1" max="6" placeholder="Credits" value="${c.credits||''}" onchange="updateCourse(${sem.id},${ci},'credits',this.value)"/>
            <button class="btn-danger" onclick="removeCourse(${sem.id},${ci})">×</button>
          </div>`).join('')}
        <div class="actions">
          <button class="btn btn-ghost btn-sm" onclick="addCourse(${sem.id})">+ Add Course</button>
        </div>
        <div class="sgpa-row" id="sgpa-${sem.id}"></div>`;
      c.appendChild(d);
    });
  }

  function calculate() {
    let totalPts = 0, totalCr = 0;
    semesters.forEach(sem => {
      let pts = 0, cr = 0;
      sem.courses.forEach(c => {
        if (!c.grade || !c.credits) return;
        pts += gradePoints[c.grade] * parseFloat(c.credits);
        cr += parseFloat(c.credits);
      });
      totalPts += pts; totalCr += cr;
      const el = document.getElementById(`sgpa-${sem.id}`);
      if (el) el.innerHTML = `Semester GPA: <span class="sgpa-val">${cr > 0 ? (pts/cr).toFixed(2) : '—'}</span>`;
    });
    const cgpa = totalCr > 0 ? (totalPts / totalCr).toFixed(2) : 0;
    const cls = cgpa >= 3.7 ? 'First Class Distinction' : cgpa >= 3.0 ? 'First Class' :
                cgpa >= 2.5 ? 'Second Class Upper' : cgpa >= 2.0 ? 'Second Class Lower' : 'Pass';
    document.getElementById('results-card').style.display = 'block';
    document.getElementById('results-container').innerHTML = `
      <div class="result-box"><span class="label">CGPA</span><div class="value">${cgpa}</div><div class="sub">out of 4.00</div></div>
      <div class="result-box"><span class="label">Total Credits</span><div class="value">${totalCr}</div><div class="sub">credit hours</div></div>
      <div class="result-box"><span class="label">Classification</span><div class="value" style="font-size:1rem;margin-top:6px;">${cls}</div></div>`;
    document.getElementById('results-card').scrollIntoView({ behavior: 'smooth' });
  }

  addSemester();
</script>
