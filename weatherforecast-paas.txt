STEPS FOR VERCEL:
1. make index.html file and paste code 
2. make repo on github, upload index.html there
3. login to vercel with github
4. add new project on vercel
5. select the github repo
6. click deploy


STEPS FOR AWS AMPLIFY:
1. make a repository on git my-paas-app and paste index.html code
2. sign in into aws amplify and use root email.
3. use code in phone to login.
4. near search bar, click on squares then aws amplify
5. click create new app
6. host using git
7. select the repository
8. then just do next next then save and deploy
9. See the deployment

<script>
  const days = ['Sunday','Monday','Tuesday','Wednesday','Thursday','Friday','Saturday'];
  const short = ['Sun','Mon','Tue','Wed','Thu','Fri','Sat'];

  // Seeded random so same city always gives same "weather"
  function seed(str) {
    let h = 0;
    for (let i = 0; i < str.length; i++) h = Math.imul(31, h) + str.charCodeAt(i) | 0;
    return function() { h ^= h << 13; h ^= h >> 17; h ^= h << 5; return (h >>> 0) / 4294967296; };
  }

  const conditions = [
    { label:'Sunny', icon:'☀️', bg:'linear-gradient(160deg,#0f2d5e,#1a6bd4,#60a5fa)', tempMod: 4 },
    { label:'Partly Cloudy',icon:'⛅', bg:'linear-gradient(160deg,#1e3a5f,#2d5f8f,#5588bb)', tempMod: 0 },
    { label:'Overcast', icon:'☁️', bg:'linear-gradient(160deg,#2d3748,#4a5568,#718096)', tempMod:-2 },
    { label:'Light Rain', icon:'🌧️', bg:'linear-gradient(160deg,#1a2030,#2d3748,#4a5568)', tempMod:-4 },
    { label:'Thunderstorm', icon:'⛈️', bg:'linear-gradient(160deg,#0d1117,#1a2030,#2d3748)', tempMod:-6 },
    { label:'Foggy', icon:'🌫️', bg:'linear-gradient(160deg,#374151,#4b5563,#6b7280)', tempMod:-1 },
    { label:'Light Snow', icon:'🌨️', bg:'linear-gradient(160deg,#1e3a5f,#2c4a7c,#3a6199)', tempMod:-10 },
    { label:'Drizzle', icon:'🌦️', bg:'linear-gradient(160deg,#1e3040,#2d4a60,#3d6a80)', tempMod:-3 },
    { label:'Clear Night', icon:'🌙', bg:'linear-gradient(160deg,#0a0f1e,#1a1f3e,#252a5e)', tempMod:-2 },
    { label:'Windy', icon:'💨', bg:'linear-gradient(160deg,#0f2040,#1a3a60,#2a5a80)', tempMod:-1 },
  ];

  // Rough base temps by city name hash (simulates geography)
  function baseTemp(rng) { return Math.round(10 + rng() * 25); }

  function humidity(rng) { return Math.round(30 + rng() * 60); }
  function wind(rng) { return Math.round(5 + rng() * 35); }
  function vis(rng) { return Math.round(5 + rng() * 15); }

  function go() {
    const city = document.getElementById('city-input').value.trim();
    if (!city) return;

    const rng = seed(city.toLowerCase());
    const todayIdx = new Date().getDay();
    const base = baseTemp(rng);
    const condIdx = Math.floor(rng() * conditions.length);
    const cond = conditions[condIdx];
    const temp = base + cond.tempMod;
    const hum = humidity(rng);
    const spd = wind(rng);
    const vis_km = vis(rng);
    const feelsLike= temp - Math.round(rng() * 4);

    document.body.style.background = cond.bg;

    // 7-day forecast
    const fcCards = Array.from({length:7}, (_, i) => {
      const r2 = seed(city.toLowerCase() + i);
      const ci = Math.floor(r2() * conditions.length);
      const fc = conditions[ci];
      const hi = base + fc.tempMod + Math.round(r2() * 4);
      const lo = hi - 4 - Math.round(r2() * 6);
      const dIdx = (todayIdx + i) % 7;
      return `<div class="fc-card">
        <div class="day">${i===0?'Today':short[dIdx]}</div>
        <span class="icon">${fc.icon}</span>
        <div class="hi">${hi}°</div>
        <div class="lo">${lo}°</div>
      </div>`;
    }).join('');

    document.getElementById('output').innerHTML = `
      <div class="main-card">
        <div class="loc-label">Current Weather</div>
        <div class="city-name">${city.split(' ').map(w=>w[0].toUpperCase()+w.slice(1)).join(' ')}</div>
        <span class="weather-icon">${cond.icon}</span>
        <div class="temp-main">${temp}°C</div>
        <div class="weather-desc">${cond.label}</div>
        <div class="stats-row">
          <div class="stat"><span class="lbl">Humidity</span><div class="val">${hum}%</div></div>
          <div class="stat"><span class="lbl">Wind</span><div class="val">${spd} km/h</div></div>
          <div class="stat"><span class="lbl">Feels Like</span><div class="val">${feelsLike}°C</div></div>
          <div class="stat"><span class="lbl">Visibility</span><div class="val">${vis_km} km</div></div>
        </div>
      </div>
      <div class="forecast-row">${fcCards}</div>
      <div class="note">Weather data is simulated using a deterministic algorithm. The same city always returns the same forecast pattern.</div>`;
  }
</script>
